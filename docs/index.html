Quick Start

# Clone the repository
git clone https://github.com/noktirnal42/openjeeves.git
cd openjeeves

# Install dependencies
pnpm install

# Build UI and backend
pnpm ui:build
pnpm build

# Run the gateway
pnpm openclaw gateway run --bind loopback --port 18789
